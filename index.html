🎮 لعبة Catch the Dot

لعبة على شاشة LCD 16x2 مع تحكم Joystick — Arduino Uno R3

💡 نظرة عامة على المشروع

لعبة تفاعلية كاملة على شاشة LCD 16x2. تتحكم بشخصية عبر Joystick، وتحاول التقاط أكبر عدد من النقاط. كل نقطة تلتقطها تعطيك +1، وإذا فاتتك 3 نقاط تنتهي اللعبة 💀

✅ كل القطع موجودة في الـ Ultimate Arduino Kit ما تحتاج تشتري شي إضافي!

🎯 شكل الشاشة:

5 ● L:3 █

● النقطة (Dot) — █ الشخصية (Player) — L:3 عدد الأرواح

💻 الكود الكامل

📋 طريقة النسخ اضغط زر "نسخ الكود" تحت، بعدها افتح Arduino IDE والصق الكود. 
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// ===== إعداد الشاشة =====
LiquidCrystal_I2C lcd(0x27, 16, 2);  // إذا ما اشتغلت غيرها إلى 0x3F

// ===== المنافذ =====
const int joyX = A0;
const int joyY = A1;
const int joySW = 7;
const int buzzerPin = 8;

// ===== إعدادات اللعبة =====
const int playerRow = 0;
const int dotRow = 1;
int playerPos = 8;
int dotPos = 0;
int score = 0;
int lives = 3;

// ===== رموز مخصصة =====
byte playerChar[8] = {
  B01110, B01110, B01110, B01110,
  B01110, B01110, B01110, B01110
};

byte dotChar[8] = {
  B00000, B00100, B01110, B11111,
  B11111, B01110, B00100, B00000
};

void setup() {
  pinMode(joySW, INPUT_PULLUP);
  pinMode(buzzerPin, OUTPUT);

  lcd.init();
  lcd.backlight();

  lcd.createChar(0, playerChar);
  lcd.createChar(1, dotChar);

  randomSeed(analogRead(A5));

  showStartScreen();
  waitForStart();
  startGame();
}

void loop() {
  // اللعبة تعمل داخل startGame
}

// ===== شاشة البداية =====
void showStartScreen() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("  Catch the Dot ");
  lcd.setCursor(0, 1);
  lcd.print("  Press Button  ");
}

void waitForStart() {
  while (digitalRead(joySW) == HIGH) {
    delay(50);
  }
  delay(300);
}

// ===== بدء اللعبة =====
void startGame() {
  score = 0;
  lives = 3;
  playerPos = 8;
  dotPos = random(16);

  while (lives > 0) {
    playRound();
  }

  gameOver();
}

// ===== جولة واحدة =====
void playRound() {
  bool caught = false;
  dotPos = random(16);

  // الصف العلوي: نقطة تظهر
  for (int row = 0; row < 2; row++) {
    int joyValue = analogRead(joyX);
    if (joyValue < 300 && playerPos > 0) playerPos--;
    if (joyValue > 700 && playerPos < 15) playerPos++;

    drawFrame(row);

    if (row == 1) {
      if (abs(playerPos - dotPos) <= 1) {
        caught = true;
        score++;
        playCatchSound();
      }
      break;
    }
    delay(500);
  }

  if (!caught) {
    lives--;
    playMissSound();
  }

  delay(300);
}

// ===== رسم الإطار =====
void drawFrame(int dotCurrentRow) {
  lcd.clear();

  if (dotCurrentRow == 0) {
    lcd.setCursor(dotPos, 0);
    lcd.write(byte(1));
  }

  lcd.setCursor(playerPos, 1);
  lcd.write(byte(0));

  lcd.setCursor(0, 0);
  lcd.print(score);
  lcd.setCursor(13, 0);
  lcd.print("L:");
  lcd.print(lives);
}

// ===== الأصوات =====
void playCatchSound() {
  tone(buzzerPin, 2000, 100);
  delay(100);
}

void playMissSound() {
  tone(buzzerPin, 300, 300);
  delay(300);
}

// ===== نهاية اللعبة =====
void gameOver() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("  GAME OVER!    ");
  lcd.setCursor(0, 1);
  lcd.print("  Score: ");
  lcd.print(score);

  for (int i = 0; i < 3; i++) {
    tone(buzzerPin, 500, 200);
    delay(250);
  }

  delay(3000);

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Press Button...");
  waitForStart();
  startGame();
}

📖 شرح مبسط للكود

🔹 الفكرة العامة

  • الشخصية (█) تتحرك على الصف السفلي (Row 1).
  • النقطة (●) تظهر عشوائياً في الصف العلوي (Row 0).
  • لما النقطة تنزل لصف الشخصية، إذا كنت قريب منها (فرق ≤ 1) → ربحت نقطة.
  • إذا ما لحقتها → تخسر روح.
  • 3 أرواح = انتهاء اللعبة.

🔹 قراءة الـ Joystick

  • analogRead(joyX) يعطي قيمة من 0 إلى 1023.
  • المنتصف ≈ 512.
  • أقل من 300 → تحرك يسار.
  • أكثر من 700 → تحرك يمين.

🔹 الرموز المخصصة

استخدمت lcd.createChar() لإنشاء شكل الشخصية والنقطة. الشاشة تسمح بـ 8 رموز مخصصة كحد أقصى.

🔹 الأصوات

  • tone(pin, freq, duration) لتشغيل نغمة على البزر.
  • صوت عالي (2000Hz) لما تلتقط نقطة.
  • صوت منخفض (300Hz) لما تخسر روح.

🛠️ خطوات التنفيذ

4

5
ارفع الكود اختر Arduino Uno والمنفذ الصحيح، واضغط Upload.
6
ابدأ اللعب! اضغط زر الـ Joystick عشان تبدأ. حرك يمين/يسار والعب!

🔧 حل المشاكل الشائعة

❗ الشاشة ما تعرض شي (أو شاشة زرقاء فاضية)

  • لف البوتنسيومتر الصغير في ظهر لوحة I2C → لتعديل السطوع.
  • غيّر العنوان من 0x27 إلى 0x3F.
  • تأكد SDA → A4 و SCL → A5.
  • تأكد VCC على 5V و GND موصول.

❗ الشخصية ما تتحرك

  • تأكد VRx على A0.
  • افتح Serial Monitor واطبع analogRead(A0) لتشوف القيم.
  • إذا القيم ثابتة على 0 أو 1023 → مشكلة توصيل.

❗ زر الـ Joystick ما يشتغل

  • تأكد SW على Pin 7.
  • الكود يستخدم INPUT_PULLUP → ما تحتاج مقاومة خارجية.

❗ البزر ما يطلع صوت

  • تأكد + على Pin 8 و − على GND.
  • البزر الفعال (Active Buzzer) يعمل مباشرة.
  • البزر السلبي (Passive) يعمل بأي تردد عبر tone().

❗ الرموز تظهر كأشكال غريبة

  • هذي مشكلة في createChar — تأكد أنك نسخت المصفوفات كاملة.
  • أعد رفع الكود.

❗ اللعبة تبدأ لحالها بدون زر

  • مشكلة في Pin 7 — تأكد أن SW موصول فيه.
  • جرب تغيّر Pin 7 إلى Pin آخر فاضي (مثل Pin 6).

✨ تحسينات ممكنة بعد ما يشتغل

  • 🎯 زيادة السرعة: قلل قيمة delay(500) كل ما زادت النقاط.
  • 🌟 مستويات: 3 مستويات بسرعات مختلفة.
  • 🎵 ألحان أفضل: أضف نغمات مختلفة للأحداث.
  • 🏆 أعلى نتيجة: احفظها في EEPROM.
  • 💡 LED: يضيء مع كل نقطة تلتقطها.
  • 🎨 أشكال شخصيات: أنشئ أشكالاً مخصصة أجمل.
  • 🔊 موسيقى خلفية: بالبزر السلبي.
  • 📱 مشاركة النتيجة: مع ESP32 لاحقاً.
